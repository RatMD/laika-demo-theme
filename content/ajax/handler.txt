function onTest(): void
{
    $value1 = (float) input('value1', 0);
    $value2 = (float) input('value2', 0);
    $operation = (string) input('operation', '*');

    $this['result'] = match ($operation) {
        '+' => $value1 + $value2,
        '-' => $value1 - $value2,
        '*' => $value1 * $value2,
        default => $value2 !== 0.0 ? round($value1 / $value2, 2) : 'NaN',
    };
}
